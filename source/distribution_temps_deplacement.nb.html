Distribution des temps de déplacements par jour et par personne —-

On ne garde que les individus dont la PCS est renseignée et on calcule la somme de leurs temps de déplacement

library(survey)

temps_quotid_personne_brut <- deplacements %>%
  group_by(NQUEST, NP) %>%
  summarise(temps = sum(DUREE)) %>%
  left_join(individus_CS8_short, by = c("NQUEST", "NP")) %>%
  mutate(temps = tidyr::replace_na(temps, 0)) %>% # O pour les individus qui ne se sont pas déplacés
  filter(!is.na(CS8))

Tableau avec les individus de toutes les PCS de niveau 1

temps_quotid_personne <- temps_quotid_personne_brut %>%
  mutate(CS8 = sjlabelled::as_label(CS8, prefix = TRUE))
temps_quotid_personne <-  sjlabelled::copy_labels(temps_quotid_personne, temps_quotid_personne_brut)

dw <- svydesign(ids = ~1,
                data = temps_quotid_personne,
                weights = ~POIDSP)
